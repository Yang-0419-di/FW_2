<script>
    // 切換顯示/隱藏任意區塊（可重複使用）
	function toggleSection(id, ev) {
		ev && ev.preventDefault && ev.preventDefault();
		const el = document.getElementById(id);
		if (!el) return;
		el.classList.toggle('section-hidden');
	}

	function toggleHUB() {
	    toggleSection('HUB-top');
		toggleSection('HUB-content');
	}
	
	// ===== 門市資料搜尋 =====
	const storeSearchBottom = document.getElementById("storeSearchBottom");
	if (storeSearchBottom) {
		storeSearchBottom.addEventListener("input", function () {
			const keyword = this.value.toLowerCase();

			document.querySelectorAll("#store tbody tr").forEach(row => {
				row.style.display =
					row.innerText.toLowerCase().includes(keyword) ? "" : "none";
			});
		});
	}

// 儲存 SC/TM 數值至 Google Sheet
function saveDiskRow(btn) {
    const tr = btn.closest('tr');
    
    // 取得 td 內容 (請根據您 HTML <td> 的實際位置調整索引)
    const tds = tr.querySelectorAll('td');
    
    const payload = {
        action: 'save',
        leave_time: tds[0].innerText.trim(),    // A欄: 離場時間
        store_id: tds[1].innerText.trim(),      // B欄: 門店編號
        store_name: tds[2].innerText.trim(),    // C欄: 門店名稱
        repair_cat: tds[3].innerText.trim(),    // D欄: 報修類別
        work_content: tds[4].innerText.trim(),  // E欄: 工作內容
        sc1: tr.querySelector('.sc1').value.trim(), // F欄: SC(1)
        sc2: tr.querySelector('.sc2').value.trim(), // G欄: SC(2)
        tm1: tr.querySelector('.tm1').value.trim(), // H欄: TM(1)
        tm2: tr.querySelector('.tm2').value.trim()  // I欄: TM(2)
    };

    fetch('/sc_disk/update', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
    })
    .then(res => res.json())
    .then(data => {
        alert(data.message);
    })
    .catch(err => alert('儲存失敗: ' + err));
}

// 刪除該筆資料（同步刪除 Google Sheet）
function deleteDiskRow(btn) {
    if (!confirm("確定要刪除此筆記錄並同步刪除 Google 試算表中的資料嗎？")) return;
    
    const tr = btn.closest('tr');
    const store_id = tr.children[1].innerText.trim();

    fetch('/sc_disk/update', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ action: 'delete', store_id: store_id })
    })
    .then(res => res.json())
    .then(data => {
        alert(data.message);
        tr.remove(); // 成功後將畫面的 Row 移除
    })
    .catch(err => alert('刪除失敗: ' + err));
}

</script>
